BRIEF OVERVIEW OF THE PROJECT

Our project, "Nine Tiles," is a desktop Tic-Tac-Toe game built using Python's Tkinter GUI framework, offering a user-friendly menu-driven interface with the following features:

User Authentication: Secure login and signup functionality using usernames and passwords, backed by a MySQL database, to control access to the game.

Multi-Difficulty Gameplay: Three distinct game modes — Easy, Medium, and Hard — each powered by a different AI strategy, letting players choose their level of challenge.

AI Opponent Logic: The Easy mode uses random move selection, the Medium mode uses rule-based heuristics (win/block detection with positional strategy), and the Hard mode uses the Minimax algorithm to play a mathematically unbeatable game.

Score and Statistics Tracking: Automatically records wins, losses, and ties for every user, separately for each difficulty level, persisting the data in a MySQL database.

Leaderboard Display: Presents per-mode leaderboards (Easy, Medium, Hard) showing the top-ranked players by combined wins and ties, and additionally displays the current user's own rank if they fall outside the top 3.



OBJECTIVE AND SCOPE

The primary objective of this Nine Tiles (Tic-Tac-Toe) project is to build an engaging, replayable desktop game that also demonstrates practical use of GUI programming, game-tree algorithms, and database-backed persistence. The project aims to achieve the following key objectives:

User Access Control: Implement secure user authentication (login/signup) to control access to the game, ensuring each player's statistics are tracked individually.

Difficulty-Based Gameplay: Offer three AI difficulty levels — Easy (random), Medium (rule-based heuristics), and Hard (Minimax) — to cater to players of varying skill levels.

Game Logic Correctness: Centralize win/tie detection logic in a shared, reusable module so all three modes and the Minimax search evaluate the board consistently.

Statistics Management: Track and persist per-user, per-mode win/loss/tie counts and total games played, enabling long-term progress tracking.

Competitive Ranking: Provide a per-mode leaderboard that ranks players by performance, motivating replayability and healthy competition among users.

Cross-Platform UI Reliability: Ensure the Tkinter interface (buttons, entries, focus behavior) renders and behaves consistently, particularly addressing macOS-specific rendering quirks.

import mysql.connector

# DB connection config
DB_CONFIG = {
    "host":     "localhost",

current_user = None

def set_current_user(username):

def _connect():
    mycon = mysql.connector.connect(**DB_CONFIG)
    if mycon.is_connected() == False:
        print('Error connected to MYSQL database')
    return mycon

def create_database():
    cfg = DB_CONFIG.copy()
    db_name = cfg.pop("database")
    conn = mysql.connector.connect(**cfg)
    cursor = conn.cursor()
    cursor.execute(f"CREATE DATABASE IF NOT EXISTS `{db_name}`")
    cursor.execute(f"USE `{db_name}`")

    cursor.execute("""
    CREATE TABLE IF NOT EXISTS users(
        username TEXT PRIMARY KEY,
        password TEXT,

        easy_wins    INTEGER DEFAULT 0,
        easy_losses  INTEGER DEFAULT 0,
        easy_ties    INTEGER DEFAULT 0,

        medium_losses INTEGER DEFAULT 0,
        medium_wins   INTEGER DEFAULT 0,
        medium_ties   INTEGER DEFAULT 0,

        hard_wins    INTEGER DEFAULT 0,
        hard_losses  INTEGER DEFAULT 0,
        hard_ties    INTEGER DEFAULT 0,

        games_played INTEGER DEFAULT 0
    )
    """)

    conn.commit()
    cursor.close()
    conn.close()

## TO REGISTER A NEW USER
def register_user(username, password):

    conn = _connect()
    cursor = conn.cursor()

    try:
        cursor.execute(
            "INSERT INTO users(username, password) VALUES (%s, %s)",
            (username, password)
        )
        conn.commit()
        return True

    except mysql.connector.IntegrityError:
        return False

    finally:
        cursor.close()
        conn.close()


## TO LOGIN AN EXISTING USER
def login_user(username, password):

    conn = _connect()
    cursor = conn.cursor()

    cursor.execute(
        "SELECT * FROM users WHERE username=%s AND password=%s",
        (username, password)
    )

    user = cursor.fetchone()
    cursor.close()
    conn.close()

    return user


## TO UPDATE WIN / LOSS / TIE COUNT FOR THE CURRENT USER
def add_status(mode, status):

    if not current_user:
        return

    conn = _connect()
    cursor = conn.cursor()

    cursor.execute(f"""
    UPDATE users
    SET {mode}_{status} = {mode}_{status} + 1,
        games_played = games_played + 1
    WHERE username = %s
    """, (current_user,))

    conn.commit()
    cursor.close()
    conn.close()


## TO GET TOP 3 PLAYERS FOR A GIVEN MODE
def get_top_players_by_mode(mode, limit=3):

    conn = _connect()
    cursor = conn.cursor()
    cursor.execute(f"""
    SELECT username, {mode}_wins AS wins, {mode}_ties AS ties
    FROM users
    ORDER BY {mode}_wins DESC, {mode}_ties DESC
    LIMIT %s
    """, (limit,))
    leaders = cursor.fetchall()
    cursor.close()
    conn.close()
    return leaders


## TO GET A SPECIFIC USER'S WINS/TIES FOR A GIVEN MODE
def get_user_stats_by_mode(mode, username):

    conn = _connect()
    cursor = conn.cursor()
    cursor.execute(f"""
    SELECT {mode}_wins, {mode}_ties FROM users WHERE username = %s
    """, (username,))
    row = cursor.fetchone()
    cursor.close()
    conn.close()
    return row


## TO COMPUTE A USER'S RANK (COUNT OF PLAYERS SCORING HIGHER)
def get_count_above_score(mode, wins, ties):

    conn = _connect()
    cursor = conn.cursor()
    cursor.execute(f"""
    SELECT COUNT(*) FROM users
    WHERE {mode}_wins > %s OR ( {mode}_wins = %s AND {mode}_ties > %s )
    """, (wins, wins, ties))
    count = cursor.fetchone()[0]
    cursor.close()
    conn.close()
    return count

def check_board(board_text):
    lines = [
        [board_text[0][0],board_text[0][1],board_text[0][2]],
        [board_text[1][0],board_text[1][1],board_text[1][2]],
        [board_text[2][0],board_text[2][1],board_text[2][2]],
        [board_text[0][0],board_text[1][0],board_text[2][0]],
        [board_text[0][1],board_text[1][1],board_text[2][1]],
        [board_text[0][2],board_text[1][2],board_text[2][2]],
        [board_text[0][0],board_text[1][1],board_text[2][2]],
        [board_text[0][2],board_text[1][1],board_text[2][0]],
    ]
    for line in lines:
        if line == ["X","X","X"]: return "X"
        if line == ["O","O","O"]: return "O"
    if all(board_text[r][c] != "" for r in range(3) for c in range(3)): return "tie"
    return None

def check_winner(board, mode):

    return 0

## MINIMAX: O IS MAXIMISER (+10), X IS MINIMISER (-10), TIE IS 0
def minimax(board, is_maximizing):
    result = check_board(board)
    if result=="O":   return 10
    if result=="X":   return -10
    if result=="tie": return 0
    if is_maximizing:
        best = -100
        for r in range(3):
            for c in range(3):
                if board[r][c]=="":
                    board[r][c]="O"
                    best = max(best, minimax(board, False))
                    board[r][c]=""
        return best
    else:
        best = 100
        for r in range(3):
            for c in range(3):
                if board[r][c]=="":
                    board[r][c]="X"
                    best = min(best, minimax(board, True))
                    board[r][c]=""
        return best

## FILL A LEADERBOARD TABLE WITH TOP 3 + CURRENT USER'S RANK
def fill_table(table, mode):
    top3 = get_top_players_by_mode(mode, limit=3)
    for row in table.get_children():
        table.delete(row)
    for i, (name, wins, ties) in enumerate(top3, start=1):
        table.insert("", tk.END, values=(i, name, wins, ties))
    me = user_module.current_user
    my_wins, my_ties = get_user_stats_by_mode(mode, me)
    my_rank = get_count_above_score(mode, my_wins, my_ties) + 1
    table.insert("", tk.END, values=(my_rank, f"{me} ★", my_wins, my_ties))
